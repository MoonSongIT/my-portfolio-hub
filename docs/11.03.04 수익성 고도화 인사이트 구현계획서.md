## 1. 개요

### 1.1 배경 및 현황

Phase 11.03에서 매매 심리 카테고리 구조를 12개(매수) + 11개(매도)로 고도화했다. 이제 카테고리 데이터가 풍부해진 상태에서 그 데이터를 **수익률 극대화 인사이트**로 전환해야 한다.

**현재 가능한 것 (기반 완료):**
- 거래별 심리 카테고리 기록 ✅
- `calcPsychologyStats()` — 심리별 평균 손익·승률 ✅
- `findRepeatedMistakes()` — 반복 손실 패턴 탐지 ✅
- `findBestPatterns()` — 최고 수익 패턴 추출 ✅
- Claude API 연동 (`src/api/claudeApi.js`) ✅

**현재 부족한 것 (이번 Phase 목표):**
- 심리별 수익률 인사이트를 **시각화**하여 즉시 파악하게 하는 대시보드
- 거래 직후 **AI 기반 역사 피드백** ("이 심리로 지난 N회 중 M회 손실")
- 반복 실수 패턴을 **사전 경고**하는 메커니즘
- 심리 성숙도 성장 추이 추적

### 1.2 구현 목표 (P1 기능 우선)

**신규 함수 1 — `getPsychologyHistory`**:

```javascript
// src/utils/calculator.js 추가

/**
 * 특정 심리 카테고리의 최근 N회 이력 조회
 * @returns { count, lossCount, lossRate, avgPnl, entries }
 */
export const getPsychologyHistory = (entries, psychology, action, recentN = 10) => {
  const filtered = entries
    .filter(e => e.psychology === psychology && e.action === action)
    .sort((a, b) => new Date(b.date) - new Date(a.date))
    .slice(0, recentN)

  const count = filtered.length
  if (count === 0) return { count: 0, lossCount: 0, lossRate: 0, avgPnl: 0, entries: [] }

  const lossCount = filtered.filter(e => (e.pnl ?? 0) < 0).length
  const avgPnl = Math.round(
    filtered.reduce((sum, e) => sum + (e.pnl ?? 0), 0) / count
  )

  return {
    count,
    lossCount,
    lossRate: Math.round((lossCount / count) * 100),
    avgPnl,
    entries: filtered,
  }
}
```

**신규 함수 2 — `calcPsychologyMaturityScore`**:

```javascript
/**
 * 심리 성숙도 점수 계산 (0~100)
 * 기준: 규칙 기반 거래 비율 50% + 전체 승률 50%
 */
export const calcPsychologyMaturityScore = (entries) => {
  if (entries.length < 5) return null

  const ruleBasedList = ['미래가치 투자', '분할매수 원칙', '목표가 실현', '손절 원칙', '리밸런싱']
  const ruleBasedRatio = entries.filter(e => ruleBasedList.includes(e.psychology)).length / entries.length
  const winRate = entries.filter(e => (e.pnl ?? 0) > 0).length / entries.length

  return Math.min(100, Math.max(0, Math.round(ruleBasedRatio * 50 + winRate * 50)))
}
```

**신규 함수 3 — `buildStockPsychologyMatrix`**:

```javascript
/**
 * 종목-심리 2D 매트릭스 데이터 생성
 * @returns { [ticker]: { [psychology]: { avgPnl, count, winRate } } }
 */
export const buildStockPsychologyMatrix = (entries) => {
  const matrix = {}

  for (const entry of entries) {
    const ticker = entry.ticker || '미분류'
    const psych = entry.psychology
    if (!psych) continue

    if (!matrix[ticker]) matrix[ticker] = {}
    if (!matrix[ticker][psych]) matrix[ticker][psych] = { total: 0, wins: 0, pnlSum: 0 }

    matrix[ticker][psych].total += 1
    matrix[ticker][psych].pnlSum += entry.pnl ?? 0
    if ((entry.pnl ?? 0) > 0) matrix[ticker][psych].wins += 1
  }

  for (const ticker of Object.keys(matrix)) {
    for (const psych of Object.keys(matrix[ticker])) {
      const cell = matrix[ticker][psych]
      cell.avgPnl = Math.round(cell.pnlSum / cell.total)
      cell.winRate = Math.round((cell.wins / cell.total) * 100)
      delete cell.pnlSum
      delete cell.wins
    }
  }

  return matrix
}
```

**`usePsychologyInsight` 훅**:

```javascript
// src/hooks/usePsychologyInsight.js

import { useMemo } from 'react'
import { useJournalStore } from '../store/journalStore'
import {
  calcPsychologyStats,
  findRepeatedMistakes,
  findBestPatterns,
  getPsychologyHistory,
  calcPsychologyMaturityScore,
  buildStockPsychologyMatrix,
} from '../utils/calculator'

export const usePsychologyInsight = () => {
  const entries = useJournalStore(s => s.entries)

  const stats = useMemo(() => calcPsychologyStats(entries), [entries])
  const mistakes = useMemo(() => findRepeatedMistakes(entries), [entries])
  const bestPatterns = useMemo(() => findBestPatterns(entries), [entries])
  const maturityScore = useMemo(() => calcPsychologyMaturityScore(entries), [entries])
  const matrix = useMemo(() => buildStockPsychologyMatrix(entries), [entries])

  const getHistory = (psychology, action) =>
    getPsychologyHistory(entries, psychology, action)

  return { stats, mistakes, bestPatterns, maturityScore, matrix, getHistory }
}
```

**`generatePostTradeCoaching` 함수**:

```javascript
// src/api/claudeApi.js 추가

/**
 * 거래 직후 코칭 메시지 생성 (100자 이내)
 * @param {object} entry  - 방금 저장된 거래 엔트리
 * @param {object} history - getPsychologyHistory 반환값
 * @param {string} apiKey
 * @returns {Promise<string>} 코칭 메시지
 */
export const generatePostTradeCoaching = async (entry, history, apiKey) => {
  const { count, lossCount, lossRate, avgPnl } = history

  const prompt = `투자 심리 코치로서, 다음 거래 기록에 대해 짧고 실용적인 피드백을 제공해줘.

[방금 기록한 거래]
- 매매: ${entry.action === 'buy' ? '매수' : '매도'}
- 심리: ${entry.psychology}

[이 심리의 과거 이력 (최근 ${count}회)]
- 손실 횟수: ${lossCount}/${count}회 (${lossRate}%)
- 평균 손익: ${avgPnl?.toLocaleString()}원

조건: 100자 이내, 데이터 기반, 공감적 어조, 투자 권유 금지`

  return callClaude(prompt, apiKey, { maxTokens: 200 })
}
```

**`PostTradeInsight` 컴포넌트**:

```jsx
// src/components/journal/PostTradeInsight.jsx

import { useEffect, useState } from 'react'
import { useJournalStore } from '../../store/journalStore'
import { generatePostTradeCoaching } from '../../api/claudeApi'
import { getPsychologyHistory } from '../../utils/calculator'
import { formatKRW } from '../../utils/formatters'

export default function PostTradeInsight({ entry, onClose }) {
  const entries = useJournalStore(s => s.entries)
  const apiKey = useJournalStore(s => s.aiApiKey)
  const [coaching, setCoaching] = useState('')
  const [loading, setLoading] = useState(false)

  const history = getPsychologyHistory(entries, entry?.psychology, entry?.action)

  useEffect(() => {
    if (!apiKey || !entry?.psychology || history.count < 2) return
    setLoading(true)
    generatePostTradeCoaching(entry, history, apiKey)
      .then(setCoaching)
      .catch(() => setCoaching(''))
      .finally(() => setLoading(false))
  }, [entry?.id])

  if (!entry) return null

  return (
    <div className="mt-4 p-4 bg-blue-50 dark:bg-blue-900/20 border border-blue-200 dark:border-blue-700 rounded-xl">
      <div className="flex items-center justify-between mb-3">
        <h4 className="text-sm font-semibold text-blue-800 dark:text-blue-200">
          💡 이 심리로 지난 거래 분석
        </h4>
        <button onClick={onClose} className="text-gray-400 hover:text-gray-600 text-xs">×</button>
      </div>

      {history.count >= 2 ? (
        <div className="space-y-1 text-sm">
          <div className="flex items-center gap-2">
            <span className="font-medium text-gray-800 dark:text-gray-200">
              {entry.psychology}
            </span>
            <span className={`font-bold ${history.lossRate >= 60 ? 'text-blue-600' : 'text-red-500'}`}>
              {history.count}회 중 {history.lossCount}회 손실 ({history.lossRate}%)
            </span>
          </div>
          <div className="text-xs text-gray-500">평균 손익: {formatKRW(history.avgPnl)}</div>
        </div>
      ) : (
        <p className="text-xs text-gray-500">첫 기록입니다. 데이터가 쌓이면 패턴 분석이 가능해집니다.</p>
      )}

      {(loading || coaching) && (
        <div className="mt-3 pt-3 border-t border-blue-200 dark:border-blue-700 flex items-start gap-2">
          <span className="text-sm">🤖</span>
          {loading
            ? <span className="text-xs text-gray-400 animate-pulse">AI 코칭 중...</span>
            : <p className="text-xs text-gray-700 dark:text-gray-300">{coaching}</p>
          }
        </div>
      )}
    </div>
  )
}
```

```jsx
// src/components/journal/PatternWarning.jsx

import { useState } from 'react'
import { useJournalStore } from '../../store/journalStore'
import { selectPatternAlerts } from '../../store/journalStore'

export default function PatternWarning() {
  const alerts = useJournalStore(selectPatternAlerts)
  const [dismissed, setDismissed] = useState([])
  const visible = alerts.filter(a => !dismissed.includes(a.psychology))

  if (!visible.length) return null

  return (
    <div className="space-y-2 mb-4">
      {visible.map(alert => (
        <div
          key={alert.psychology}
          className={`flex items-start justify-between p-3 rounded-lg border text-sm ${
            alert.level === 'danger'
              ? 'bg-red-50 dark:bg-red-900/20 border-red-300 dark:border-red-700'
              : 'bg-yellow-50 dark:bg-yellow-900/20 border-yellow-300 dark:border-yellow-700'
          }`}
        >
          <div>
            <span className="mr-1">{alert.level === 'danger' ? '🔴' : '🟡'}</span>
            <span className="font-semibold">{alert.psychology}</span>
            <span className="ml-2 text-gray-600 dark:text-gray-400 text-xs">
              최근 {alert.totalCount}회 중 {alert.lossCount}회 손실 ({alert.lossRate}%) |
              평균 손실 {Math.round(Math.abs(alert.avgLoss) / 10000).toLocaleString()}만원
            </span>
          </div>
          <button
            onClick={() => setDismissed(prev => [...prev, alert.psychology])}
            className="text-gray-400 hover:text-gray-600 ml-2 shrink-0 text-xs"
            aria-label="경고 닫기"
          >
            ×
          </button>
        </div>
      ))}
    </div>
  )
}
```
