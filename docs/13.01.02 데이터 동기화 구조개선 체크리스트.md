  ```sql
  CREATE TABLE profiles (
    id              UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email           TEXT NOT NULL UNIQUE,
    display_name    TEXT DEFAULT '',
    created_at      TIMESTAMPTZ DEFAULT NOW(),
    updated_at      TIMESTAMPTZ DEFAULT NOW()
  ```

  ```sql

  UNION ALL

  UNION ALL
  SELECT 'cash_flows', COUNT(*) FROM cash_flows WHERE user_id = auth.uid()
  UNION ALL
  SELECT 'watchlist', COUNT(*) FROM watchlist WHERE user_id = auth.uid()
  UNION ALL
  SELECT 'calendar_events', COUNT(*) FROM calendar_events WHERE user_id = auth.uid();
  ```
