## 1. 진입점 — AIChat.jsx

**파일:** `src/pages/AIChat.jsx`

사용자가 메시지를 입력하고 전송 버튼을 누르면 `handleSend()` 함수가 실행된다.

```js
// AIChat.jsx L107~154
async function handleSend(text) {
  const msg = text?.trim()
  if (!msg || isLoading || !isOnline) return
  const ok = await ensureKey()   // API 키 미등록 시 다이얼로그 표시
  if (!ok) return

  addUserMessage(msg)            // chatStore에 사용자 메시지 추가
  setLocalLoading(true)

  try {
    // ① 10턴 초과 시 이전 대화 요약 압축
    let messagesToSend = messages.map(m => ({ role: m.role, content: m.content }))
    if (messagesToSend.length > 20) {
      messagesToSend = await summarizeAndCompressHistory(messagesToSend, systemPrompt)
    }

    // ② AIChat 자체 라우팅 보강 (오타·표현 다양성 대응)
    const forceAgent = (mentionsPortfolio && mentionsMarketEvent && context.holdings?.length > 0)
      ? 'analysis'
      : null

    // ③ 에이전트 호출
    const result = await sendToAgent(msg, context, forceAgent)

    addAIMessage(result.text, result.agentType, result.agentInfo)
    await saveCurrentSession(currentUser.id)   // IndexedDB 저장
  } catch (err) { ... }
}
```

**context 조립** (L65~78): 에이전트에 전달할 로컬 데이터를 사전에 묶음.

```js
const context = useMemo(() => ({
  holdings:      holdings.map(h => ({ ticker, name, quantity, avgPrice, currentPrice, market })),
  exchangeRate,
  watchlist:     watchlist.map(w => ({ ticker, name, market })),
  journalEntries,
  accounts,
}), [holdings, exchangeRate, watchlist, journalEntries, accounts])
```

> 주의: `holdings`에는 `changePercent`(당일 등락률)가 **없다.** analysis 에이전트에서 별도로 실시간 조회한다.

## 2. 라우팅 — orchestrator.js

**파일:** `src/agents/orchestrator.js`

### 2-1. routeToAgent() — 키워드 매칭

```js
// orchestrator.js L40~60
export function routeToAgent(userMessage) {
  const msg = userMessage.toLowerCase()

  // compound 조건 최우선: 포트폴리오 + 시장 이벤트 키워드 동시 포함
  const hasPortfolio    = msg.includes('포트폴리오') || msg.includes('내 종목') || msg.includes('보유')
  const hasMarketEvent  = msg.includes('뉴스') || msg.includes('시장') || msg.includes('등락')
                          || msg.includes('급등') || msg.includes('급락')
  if (hasPortfolio && hasMarketEvent) return 'analysis'

  // ROUTING_RULES 순서대로 첫 매칭 에이전트 반환
  for (const rule of ROUTING_RULES) {
    for (const keyword of rule.keywords) {
      if (msg.includes(keyword.toLowerCase())) return rule.agent
    }
  }
  return 'portfolio'  // fallback
}
```

## 3. 에이전트 실행 — claudeApi.js

**파일:** `src/api/claudeApi.js`

`sendToAgent()`는 라우팅 결과에 따라 컨텍스트를 빌드하고 Claude API를 호출한다.

```js
// claudeApi.js L107~254
export async function sendToAgent(userMessage, context = {}, forceAgent = null) {
  const agentType = forceAgent || routeToAgent(userMessage)
  const maxTokens = AGENT_MAX_TOKENS[agentType]   // 에이전트별 차등 적용

  let contextText = ''
  let systemPrompt = AGENT_PROMPTS[agentType]

  switch (agentType) {
    case 'analysis': { ... }   // ← 4절에서 상세 설명
    case 'journal':  { ... }
    case 'research': { ... }
    case 'portfolio': { ... }
    case 'alert':     { ... }
    case 'report':    { ... }
  }

  const combinedMessage = contextText
    ? `${contextText}\n\n사용자 질문: ${userMessage}`
    : userMessage

  // 최대 3회 시도 (500/529 에러 시 재시도, 429/401은 즉시 중단)
  for (let attempt = 0; attempt < 3; attempt++) {
    const response = await claudeApi.post('/claude', {
      systemPrompt,
      messages: [{ role: 'user', content: combinedMessage }],
      maxTokens,
    })
    return { text: response.data?.content?.[0]?.text, agentType, agentInfo }
  }
}
```

**max_tokens 설정:**

```js
// claudeApi.js L62~69
const AGENT_MAX_TOKENS = {
  analysis: 2048,
  journal:  4096,
  report:   4096,
  research: 3072,
  portfolio: 2048,
  alert:     2048,
}
```

## 4. analysis 에이전트 3-경로 분기

`analysis` 선택 시 context 내용에 따라 세 가지 다른 경로를 탄다.

```js
// claudeApi.js L119~177
case 'analysis': {
  const sd     = context.researchBundle?.stockData ?? context.stockData ?? null
  const ticker = sd?.symbol ?? sd?.ticker ?? null

  if (ticker) {
    // ─── 경로 1: 종목 급등락 원인 분석 ─────────────────────────────
    const [news, disclosures] = await Promise.all([
      fetchNews(ticker, market).catch(() => []),
      fetchDisclosures(ticker, market).catch(() => []),
    ])
    contextText  = buildMovementContext({ ticker, name, changePercent, market, news, disclosures })
    systemPrompt = ANALYSIS_PROMPT

  } else if (context.holdings?.length > 0) {
    // ─── 경로 2: 포트폴리오 + 시장 뉴스 연관 분석 ──────────────────
    const [enrichedHoldings, news] = await Promise.all([
      Promise.all(
        context.holdings.map(async (h) => {
          const q = await fetchQuote(h.ticker, h.market)   // 각 종목 quote 병렬 조회
          return { ...h, changePercent: q?.changePercent ?? null }
        })
      ),
      fetchNews('^GSPC', 'NYSE').catch(() => []),
    ])
    contextText  = buildPortfolioMovementContext({ holdings: enrichedHoldings, news })
    systemPrompt = PORTFOLIO_ANALYSIS_PROMPT

  } else {
    // ─── 경로 3: 전체 시장 브리핑 ────────────────────────────────────
    const BRIEF_INDICES = [
      { label: 'KOSPI',  ticker: '^KS11' },
      { label: 'KOSDAQ', ticker: '^KQ11' },
      { label: 'NASDAQ', ticker: '^IXIC' },
      { label: 'S&P500', ticker: '^GSPC' },
    ]
    const [quotes, news] = await Promise.all([
      Promise.all(BRIEF_INDICES.map(idx => fetchQuote(idx.ticker, 'NYSE').catch(() => null))),
      fetchNews('^GSPC', 'NYSE').catch(() => []),
    ])
    contextText  = buildMarketBriefContext({ indices, news })
    systemPrompt = MARKET_BRIEF_PROMPT
  }
}
```

export async function fetchDisclosures(ticker, market, days = 30) {
  if (isKorean) {
    // DART OpenAPI: /api/dart/list
    const res = await api.get('/dart/list', { params: { ticker, days }, headers })

// claudeApi.js L285~319
export async function summarizeAndCompressHistory(messages, systemPrompt) {
  if (messages.length <= 20) return messages  // 10턴 이하는 그대로

  const toSummarize = messages.slice(0, -4)   // 최근 4개 제외한 이전 대화
  const recent      = messages.slice(-4)

  // Claude에게 요약 요청 (max_tokens: 512)
  const summary = await claudeApi.post('/claude', {
    messages: [{ role: 'user', content: summaryPrompt }],
    maxTokens: 512,
  })

  return [
    { role: 'user',      content: `[이전 대화 요약]\n${summary}` },
    { role: 'assistant', content: '이전 대화 내용을 확인했습니다.' },
    ...recent,
  ]
}
```

## 8. 전체 호출 흐름 요약

```
AIChat.jsx
  └─ handleSend()  [L107]
       ├─ summarizeAndCompressHistory()  [10턴 초과 시]
       ├─ forceAgent 판별  [L130~141]
       └─ sendToAgent(msg, context, forceAgent)  [claudeApi.js L107]

            │    ├─ 'analysis'

            │    ├─ 'alert'   → buildAlertContext()
            │    └─ 'report'  → buildReportContext()
